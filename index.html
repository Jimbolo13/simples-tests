Un seul coupable.") # ---------- LOCATIONS ---------- def library(state): divider() slow_print("Vous entrez dans la bibliothèque poussiéreuse...") if "letter" not in state.clues: slow_print("Vous trouvez une lettre cachée dans un livre.") slow_print("Indice obtenu : Lettre mentionnant un conflit d'héritage.") state.clues.append("letter") else: slow_print("Rien de nouveau ici.") def garden(state): divider() slow_print("Le jardin est calme, presque trop...") if "footprints" not in state.clues: slow_print("Des empreintes fraîches près de la fontaine.") slow_print("Indice obtenu : Quelqu'un a menti sur sa présence.") state.clues.append("footprints") else: slow_print("Le vent souffle doucement. Explorer la bibliothèque") print("2. Explorer le jardin") print("3. Explorer la cave") print("4. Interroger un suspect") print("5. Accuser") choice = input("> ") if choice == "1": library(state) elif choice == "2": garden(state) elif choice == "3": cellar(state) elif choice == "4": interrogate(state) elif choice == "5": accuse(state) else: slow_print("Choix invalide.") if __name__ == "__main__": main()
